{{- /* Overrides Hugo's built-in figure shortcode so that figures match the theme's markdown image render hook: same .image-caption styling and the same CSS-only zoom when params.imageZoom is on. Params: src, alt, caption, title, width, height, link, target, rel, class, attr, attrlink. */ -}} {{- $src := .Get "src" -}} {{- $alt := .Get "alt" | default (.Get "caption" | markdownify | plainify) -}} {{- /* Authors write width="300px" as often as width="300"; a CSS length has to go through style, a bare number can stay an attribute. */ -}} {{- $w := .Get "width" -}} {{- $h := .Get "height" -}} {{- $style := "" -}} {{- if and $w (not (findRE `^[0-9]+$` $w)) -}} {{- $style = printf "max-width:%s;" $w -}} {{- $w = "" -}} {{- end -}} {{- if and $h (not (findRE `^[0-9]+$` $h)) -}} {{- $style = printf "%smax-height:%s;" $style $h -}} {{- $h = "" -}} {{- end -}} {{- $zoom := and $.Page.Site.Params.imageZoom (not (.Get "link")) -}} {{- $id := printf "img-%s" (substr (md5 (printf "%s-%s" $src (.Get "caption"))) 0 8) -}}
